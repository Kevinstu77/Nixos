clone github repo

nix-shell -p git --run "git clone https://github.com/Kevinstu77/Nixos.git /tmp/Nixproject"

use disko

sudo nix run github:nix-community/disko -- --mode disko /tmp/Nixproject/<your-disko-file>.nix

generate hardware conf

sudo nixos-generate-config --no-filesystems --root /mnt

make new host

mkdir -p /tmp/Nixproject/Hosts/NewLaptopName
cp /mnt/etc/nixos/hardware-configuration.nix /tmp/Nixproject/Hosts/NewLaptopName/
cp /tmp/Nixproject/Hosts/Lenovo-Slim-Pro-7/configuration.nix /tmp/Nixproject/Hosts/NewLaptopName/configuration.nix
# edit configuration.nix: hostname, GPU modules, etc.
